\chapter{PEMBAHASAN}
\section{Tinjauan Institusi/Perusahaan}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}
\textit{\textbf{Pedoman Penulisan Tinjauan Institusi/Perusahaan:} Uraikan profil lengkap tempat PKL/Magang, termasuk bidang usaha, visi-misi, serta lokasi kantor atau tempat pelaksanaan magang.}
\end{minipage}}
\vspace{0.5cm}

\subsection{Sejarah Institusi/Perusahaan}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}
\textit{\textbf{Pedoman Penulisan Sejarah Institusi/Perusahaan:} Tuliskan sejarah singkat berdirinya instansi atau perusahaan, perkembangan utama dari masa ke masa, hingga kondisi saat ini.}
\end{minipage}}
\vspace{0.5cm}

\subsection{Struktur Organisasi dan Fungsi}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}
\textit{\textbf{Pedoman Penulisan Struktur Organisasi dan Fungsi:} Tampilkan bagan/gambar struktur organisasi perusahaan, serta uraikan tugas pokok dan fungsi (job description) dari masing-masing divisi atau unit kerja yang ada.}
\end{minipage}}
\vspace{0.5cm}

\section{Skema Jaringan}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}
\textit{\textbf{Pedoman Penulisan Skema Jaringan:} Gambar dan jelaskan skema infrastruktur jaringan atau topologi fisik/logis yang terpasang pada instansi/perusahaan tersebut.}
\end{minipage}}
\vspace{0.5cm}

\section{Analisa Kemampuan Jaringan Berjalan}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}
\textit{\textbf{Pedoman Penulisan Analisa Kemampuan Jaringan Berjalan:} Lakukan analisis menyeluruh terhadap kinerja sistem atau infrastruktur jaringan yang sedang digunakan saat ini. Sebutkan kelebihan, kelemahan, serta kendala-kendala teknis yang sering muncul.}
\end{minipage}}
\vspace{0.5cm}

\subsection{Spesifikasi Bentuk Dokumen Sistem Berjalan}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}
\textit{\textbf{Pedoman Penulisan Spesifikasi Bentuk Dokumen Sistem Berjalan:} Detailkan berkas-berkas administratif yang mengalir pada sistem berjalan. Dibagi menjadi dua bagian utama: Dokumen Masukan (Input) dan Dokumen Keluaran (Output), lengkap dengan nama dokumen, sumber, fungsi, media, dan frekuensinya.}
\end{minipage}}
\vspace{0.5cm}

\subsection{Spesifikasi File}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}
\textit{\textbf{Pedoman Penulisan Spesifikasi File:} Deskripsikan struktur database atau file penyimpanan data elektronik yang digunakan dalam sistem berjalan (seperti nama tabel, tipe data kolom, panjang field, dan kunci utama/foreign key).}
\end{minipage}}
\vspace{0.5cm}

\subsection{Struktur Kode}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}
\textit{\textbf{Pedoman Penulisan Struktur Kode:} Uraikan sistem pengkodean data yang digunakan pada perusahaan (seperti Kode Barang, Kode Anggota, Kode Transaksi, dll.) beserta arti digit pengkodean tersebut.}
\end{minipage}}
\vspace{0.5cm}

\subsection{Spesifikasi Program}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}
\textit{\textbf{Pedoman Penulisan Spesifikasi Program:} Detailkan spesifikasi modul program atau sistem aplikasi yang dikembangkan atau digunakan, termasuk fungsi program, bahasa pemrograman, platform runtime, dan spesifikasi antarmuka.}
\end{minipage}}
\vspace{0.5cm}
