\noindent\fbox{\begin{minipage}{\textwidth}

\end{minipage}}
\vspace{0.5cm}

\section{Maksud dan Tujuan}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}
\textit{\textbf{Pedoman Penulisan Maksud dan Tujuan:} Maksud mendeskripsikan aktivitas magang/PKL yang dilaksanakan oleh mahasiswa pada perusahaan terkait. Tujuan mendeskripsikan hal-hal konkret yang ingin dicapai melalui analisis dan solusi sistem yang diajukan.}
\end{minipage}}
\vspace{0.5cm}

\section{Metode Penelitian}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}

\end{minipage}}
\vspace{0.5cm}

\subsection{Observasi}

\vspace{0.5cm}
\noindent\fbox{\begin{minipage}{\textwidth}

\end{minipage}}
\vspace{0.5cm}

\subsection{Wawancara}
